# [Webmin](https://webmin.com/), Interface de administração baseada na web para sistemas Unix
# Acesso: 127.0.0.1:10000, https://192.168.122.182:10000
# Interessante: Cron básico mas fácil, Configuração de rede (Complexo até se acostumar), Partições, LVM, RAID, Gerenciamento de pacotes
curl -fsSL https://download.webmin.com/jcameron-key.asc | sudo gpg --dearmor -o /usr/share/keyrings/webmin.gpg
echo "deb [signed-by=/usr/share/keyrings/webmin.gpg] http://download.webmin.com/download/repository sarge contrib" | sudo tee /etc/apt/sources.list.d/webmin.list
sudo apt update && sudo apt -y install webmin --install-recommends
